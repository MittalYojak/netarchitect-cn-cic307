==================== CORE-SW1 ====================
hostname CORE-SW1
no ip domain-lookup
ip routing
!
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,50,60,70 priority 4096
!
vlan 999
 name BLACKHOLE-NATIVE
!
vlan 10
 name Students
!
vlan 20
 name Staff
!
vlan 30
 name Labs
!
vlan 40
 name Wireless-APs
!
vlan 50
 name VoIP
!
vlan 60
 name Servers
!
vlan 70
 name Management
!
interface Vlan10
 description Gateway for Students
 ip address 10.10.0.1 255.255.252.0
 ip access-group PROTECT-MGMT in
 no shutdown
!
interface Vlan20
 description Gateway for Staff
 ip address 10.10.4.1 255.255.254.0
 ip access-group PROTECT-MGMT in
 no shutdown
!
interface Vlan30
 description Gateway for Labs
 ip address 10.10.6.1 255.255.255.0
 ip access-group PROTECT-MGMT in
 no shutdown
!
interface Vlan40
 description Gateway for Wireless-APs
 ip address 10.10.7.1 255.255.255.0
 ip access-group PROTECT-MGMT in
 no shutdown
!
interface Vlan50
 description Gateway for VoIP
 ip address 10.10.8.1 255.255.255.128
 ip access-group PROTECT-MGMT in
 no shutdown
!
interface Vlan60
 description Gateway for Servers
 ip address 10.10.8.129 255.255.255.192
 ip access-group PROTECT-MGMT in
 no shutdown
!
interface Vlan70
 description Gateway for Management
 ip address 10.10.8.193 255.255.255.224
 no shutdown
!
interface GigabitEthernet0/1
 description Trunk uplink to distribution/core switch
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk allowed vlan 10,20,30,40,50,60,70
 switchport trunk native vlan 999
 switchport nonegotiate
 no shutdown
!
ip access-list extended PROTECT-MGMT
 remark Restrict access to the management subnet
 deny ip any 10.10.8.192 0.0.0.31
 permit ip any any
!
router ospf 1
 router-id 1.1.1.1
 passive-interface default
 network 10.10.0.0 0.0.3.255 area 0
 network 10.10.4.0 0.0.1.255 area 0
 network 10.10.6.0 0.0.0.255 area 0
 network 10.10.7.0 0.0.0.255 area 0
 network 10.10.8.0 0.0.0.127 area 0
 network 10.10.8.128 0.0.0.63 area 0
 network 10.10.8.192 0.0.0.31 area 0
!
end
